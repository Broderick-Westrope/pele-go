package mermaid

import (
	"context"
	"os"
	"os/exec"
	"runtime"
	"strconv"
	"strings"
	"testing"

	pele "github.com/Broderick-Westrope/pele-go"
	_ "github.com/Broderick-Westrope/pele-go/all"
)

const memSource = "info\ntitle Memory check"

func rssKB(t *testing.T) int {
	out, err := exec.Command("ps", "-o", "rss=", "-p", strconv.Itoa(os.Getpid())).Output()
	if err != nil {
		t.Fatal(err)
	}
	n, _ := strconv.Atoi(strings.TrimSpace(string(out)))
	return n
}

func settle() runtime.MemStats {
	runtime.GC()
	runtime.GC()
	var m runtime.MemStats
	runtime.ReadMemStats(&m)
	return m
}

func report(t *testing.T, label string, base runtime.MemStats, baseRSS int) {
	m := settle()
	t.Logf("%s: heap in use +%d KB, Go sys +%d KB, RSS +%d KB", label,
		(int64(m.HeapInuse)-int64(base.HeapInuse))/1024, (int64(m.Sys)-int64(base.Sys))/1024, rssKB(t)-baseRSS)
}

func TestMemGoja(t *testing.T) {
	base, baseRSS := settle(), rssKB(t)
	r, err := NewRenderer()
	if err != nil {
		t.Fatal(err)
	}
	rast, err := NewRasterizer(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	for range 20 {
		res, err := r.Render(memSource, Options{})
		if err != nil {
			t.Fatal(err)
		}
		if _, err := rast.Rasterize(res, testTheme, RasterOptions{Scale: 2}); err != nil {
			t.Fatal(err)
		}
	}
	report(t, "goja Pele + resvg, loaded", base, baseRSS)
	runtime.KeepAlive(r)
	runtime.KeepAlive(rast)
}

func TestMemGo(t *testing.T) {
	base, baseRSS := settle(), rssKB(t)
	rast, err := NewRasterizer(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	for range 20 {
		pr, err := pele.Render(context.Background(), memSource, pele.Options{})
		if err != nil {
			t.Fatal(err)
		}
		if _, err := rast.Rasterize(Result{SVG: pr.SVG, Width: pr.Width, Height: pr.Height}, testTheme, RasterOptions{Scale: 2}); err != nil {
			t.Fatal(err)
		}
	}
	report(t, "pele-go + resvg, loaded", base, baseRSS)
	runtime.KeepAlive(rast)
}
