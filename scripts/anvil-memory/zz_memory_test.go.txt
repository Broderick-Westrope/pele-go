package mermaid

import (
	"context"
	"os"
	"os/exec"
	"runtime"
	"strconv"
	"strings"
	"testing"

	pele "github.com/Broderick-Westrope/pele-go"
	_ "github.com/Broderick-Westrope/pele-go/all"
)

const memSource = "info\ntitle Memory check"

// dejaVu lets pele-go measure with Anvil's own measurer, the DejaVu Sans faces resvg draws
// with, so both paths lay out the same labels.
type dejaVu struct{ m *measurer }

func (d dejaVu) Width(text string, size float64, style pele.Style) float64 {
	return d.m.width(text, size, int(style))
}

func rssKB(t *testing.T) int {
	out, err := exec.Command("ps", "-o", "rss=", "-p", strconv.Itoa(os.Getpid())).Output()
	if err != nil {
		t.Fatal(err)
	}
	n, _ := strconv.Atoi(strings.TrimSpace(string(out)))
	return n
}

func settle() runtime.MemStats {
	runtime.GC()
	runtime.GC()
	var m runtime.MemStats
	runtime.ReadMemStats(&m)
	return m
}

func report(t *testing.T, label string, base runtime.MemStats, baseRSS int) {
	m := settle()
	t.Logf("%s: heap in use +%d KB, Go sys +%d KB, RSS +%d KB", label,
		(int64(m.HeapInuse)-int64(base.HeapInuse))/1024, (int64(m.Sys)-int64(base.Sys))/1024, rssKB(t)-baseRSS)
}

func TestMemGoja(t *testing.T) {
	base, baseRSS := settle(), rssKB(t)
	r, err := NewRenderer()
	if err != nil {
		t.Fatal(err)
	}
	rast, err := NewRasterizer(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	// The last image is kept, as the chat keeps the one it shows.
	var shown any
	for range 20 {
		res, err := r.Render(memSource, Options{})
		if err != nil {
			t.Fatal(err)
		}
		if shown, err = rast.Rasterize(res, testTheme, RasterOptions{Scale: 2}); err != nil {
			t.Fatal(err)
		}
	}
	report(t, "goja Pele + resvg, loaded", base, baseRSS)
	runtime.KeepAlive(r)
	runtime.KeepAlive(rast)
	runtime.KeepAlive(shown)
}

func TestMemGo(t *testing.T) {
	base, baseRSS := settle(), rssKB(t)
	m, err := newMeasurer()
	if err != nil {
		t.Fatal(err)
	}
	rast, err := NewRasterizer(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	// The same options Anvil's goja renderer passes: no responsive sizing, the DejaVu measurer.
	opts := pele.Options{Measurer: dejaVu{m}, Responsive: new(bool)}
	var shown any
	for range 20 {
		pr, err := pele.Render(context.Background(), memSource, opts)
		if err != nil {
			t.Fatal(err)
		}
		if shown, err = rast.Rasterize(Result{SVG: pr.SVG, Width: pr.Width, Height: pr.Height}, testTheme, RasterOptions{Scale: 2}); err != nil {
			t.Fatal(err)
		}
	}
	report(t, "pele-go + resvg, loaded", base, baseRSS)
	runtime.KeepAlive(m)
	runtime.KeepAlive(rast)
	runtime.KeepAlive(shown)
}
